// ==========================================================
// WINFUT — L1/L2 0,68% + Fibonacci (17:00 do dia anterior)
//
// Referencia: fechamento do candle das 17:00 de ontem.
//   L1 = ajuste + 0,68%   (acima, vermelho)
//   L2 = ajuste - 0,68%   (abaixo, verde)
//
// Fibonacci (L1 = 0%, L2 = 38,2%):
//   mais 3 linhas ABAIXO: 50%, 61,8% e 100%   (verde)
// Invertendo L1 e L2 (L2 = 0%, L1 = 38,2%):
//   mais 3 linhas ACIMA:  50%, 61,8% e 100%   (vermelho)
//
// NTSL: nao usar Close[1]/Time[1]/Date[1] dentro de If.
// ==========================================================

input
  HoraAjuste(170000);
  PercL1L2(0.68);

var
  HoraRef          : Integer;
  HoraAtual        : Integer;
  HoraAnt          : Integer;
  DataAnt          : Integer;
  CloseAnt         : Float;
  Fech17hAnt       : Float;
  AjusteAnt        : Float;

  Fechamento17h    : Float;
  Ajuste           : Float;

  Perc             : Float;
  L1               : Float;
  L2               : Float;
  Unidade          : Float;

  Acima50          : Float;
  Acima618         : Float;
  Acima100         : Float;
  Abaixo50         : Float;
  Abaixo618        : Float;
  Abaixo100        : Float;

  PodePlotar       : Integer;

begin

  DataAnt := Date[1];
  HoraAnt := Time[1];
  CloseAnt := Close[1];
  Fech17hAnt := Fechamento17h[1];
  AjusteAnt := Ajuste[1];

  HoraRef := HoraAjuste;
  HoraAtual := Time;

  If HoraRef < 2400 Then
    HoraRef := HoraRef * 100;

  If HoraAtual < 2400 Then
    HoraAtual := HoraAtual * 100;

  If HoraAnt < 2400 Then
    HoraAnt := HoraAnt * 100;

  Fechamento17h := Fech17hAnt;
  Ajuste := AjusteAnt;

  If Date <> DataAnt Then
    Ajuste := Fech17hAnt;

  If Date <> DataAnt Then
    Fechamento17h := Fech17hAnt;

  If Date = DataAnt Then
    If HoraAtual = HoraRef Then
      Fechamento17h := Close;

  If Date = DataAnt Then
    If HoraAtual > HoraRef Then
      If HoraAnt < HoraRef Then
        Fechamento17h := CloseAnt;

  Perc := PercL1L2 / 100;

  L1 := 0;
  L2 := 0;
  Unidade := 0;

  If Ajuste > 0 Then
    L1 := Ajuste * (1 + Perc);

  If Ajuste > 0 Then
    L2 := Ajuste * (1 - Perc);

  If L1 > L2 Then
    Unidade := (L1 - L2) / 0.382;

  Abaixo50  := L1 - (Unidade * 0.500);
  Abaixo618 := L1 - (Unidade * 0.618);
  Abaixo100 := L1 - (Unidade * 1.000);

  Acima50   := L2 + (Unidade * 0.500);
  Acima618  := L2 + (Unidade * 0.618);
  Acima100  := L2 + (Unidade * 1.000);

  PodePlotar := 0;

  If LastBarOnChart Then
    If Ajuste > 0 Then
      If Unidade > 0 Then
        PodePlotar := 1;

  If PodePlotar = 1 Then
  Begin

    HorizontalLineCustom(L1, clred, 2, 0, "L1", 10, 1, 1241701, 0, 0);
    HorizontalLineCustom(L2, clgreen, 2, 0, "L2", 10, 1, 1241702, 0, 0);

    HorizontalLineCustom(Acima50, clred, 1, 1, "50%", 10, 1, 1241703, 0, 0);
    HorizontalLineCustom(Acima618, clred, 1, 1, "61.8%", 10, 1, 1241704, 0, 0);
    HorizontalLineCustom(Acima100, clred, 1, 1, "100%", 10, 1, 1241705, 0, 0);

    HorizontalLineCustom(Abaixo50, clgreen, 1, 1, "50%", 10, 1, 1241706, 0, 0);
    HorizontalLineCustom(Abaixo618, clgreen, 1, 1, "61.8%", 10, 1, 1241707, 0, 0);
    HorizontalLineCustom(Abaixo100, clgreen, 1, 1, "100%", 10, 1, 1241708, 0, 0);

    PlotText("17H " + Ajuste + "  L1 " + L1 + "  L2 " + L2, clwhite, 0, 6);
    PlotText("100% " + Acima100 + " / " + Abaixo100, clwhite, 0, 3);

  End;

End;
