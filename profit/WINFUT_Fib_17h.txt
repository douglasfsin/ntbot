{ WINFUT L1/L2 0.68% + Fibonacci a partir das 17:00 de ontem }

input
  HoraAjuste(170000);
  PercVariacao(0.68);

var
  HoraRef : Integer;
  HoraAtual : Integer;
  HoraAnt : Integer;
  DataAnt : Integer;
  CloseAnt : Float;
  FechAnt : Float;
  AjusteAnt : Float;
  FechamentoAjuste : Float;
  Ajuste : Float;
  Perc : Float;
  LinhaAlta : Float;
  LinhaBaixa : Float;
  Unidade : Float;
  AltaMeio : Float;
  AltaOuro : Float;
  AltaCem : Float;
  BaixaMeio : Float;
  BaixaOuro : Float;
  BaixaCem : Float;
  PodePlotar : Integer;

begin

  DataAnt := Date[1];
  HoraAnt := Time[1];
  CloseAnt := Close[1];
  FechAnt := FechamentoAjuste[1];
  AjusteAnt := Ajuste[1];

  HoraRef := HoraAjuste;
  HoraAtual := Time;

  If HoraRef < 2400 Then
    HoraRef := HoraRef * 100;

  If HoraAtual < 2400 Then
    HoraAtual := HoraAtual * 100;

  If HoraAnt < 2400 Then
    HoraAnt := HoraAnt * 100;

  FechamentoAjuste := FechAnt;
  Ajuste := AjusteAnt;

  If Date <> DataAnt Then
    Ajuste := FechAnt;

  If Date <> DataAnt Then
    FechamentoAjuste := FechAnt;

  If Date = DataAnt Then
    If HoraAtual = HoraRef Then
      FechamentoAjuste := Close;

  If Date = DataAnt Then
    If HoraAtual > HoraRef Then
      If HoraAnt < HoraRef Then
        FechamentoAjuste := CloseAnt;

  Perc := PercVariacao / 100;

  LinhaAlta := 0;
  LinhaBaixa := 0;
  Unidade := 0;

  If Ajuste > 0 Then
    LinhaAlta := Ajuste * (1 + Perc);

  If Ajuste > 0 Then
    LinhaBaixa := Ajuste * (1 - Perc);

  If LinhaAlta > LinhaBaixa Then
    Unidade := (LinhaAlta - LinhaBaixa) / 0.382;

  BaixaMeio := LinhaAlta - (Unidade * 0.500);
  BaixaOuro := LinhaAlta - (Unidade * 0.618);
  BaixaCem := LinhaAlta - (Unidade * 1.000);

  AltaMeio := LinhaBaixa + (Unidade * 0.500);
  AltaOuro := LinhaBaixa + (Unidade * 0.618);
  AltaCem := LinhaBaixa + (Unidade * 1.000);

  PodePlotar := 0;

  If LastBarOnChart Then
    If Ajuste > 0 Then
      If Unidade > 0 Then
        PodePlotar := 1;

  If PodePlotar = 1 Then
  Begin
    HorizontalLineCustom(LinhaAlta, clred, 2, 0, "L1", 10, 1, 1241701, 0, 0);
    HorizontalLineCustom(LinhaBaixa, clgreen, 2, 0, "L2", 10, 1, 1241702, 0, 0);
    HorizontalLineCustom(AltaMeio, clred, 1, 1, "50%", 10, 1, 1241703, 0, 0);
    HorizontalLineCustom(AltaOuro, clred, 1, 1, "61.8%", 10, 1, 1241704, 0, 0);
    HorizontalLineCustom(AltaCem, clred, 1, 1, "100%", 10, 1, 1241705, 0, 0);
    HorizontalLineCustom(BaixaMeio, clgreen, 1, 1, "50%", 10, 1, 1241706, 0, 0);
    HorizontalLineCustom(BaixaOuro, clgreen, 1, 1, "61.8%", 10, 1, 1241707, 0, 0);
    HorizontalLineCustom(BaixaCem, clgreen, 1, 1, "100%", 10, 1, 1241708, 0, 0);
    PlotText("17H " + Ajuste + "  L1 " + LinhaAlta + "  L2 " + LinhaBaixa, clwhite, 0, 6);
    PlotText("100% " + AltaCem + " / " + BaixaCem, clwhite, 0, 3);
  End;

End;
