   Usuário:  pos
   Sistema:  Exibe tabela com todas as posições abertas
             Mostra gain/loss de cada posição
             Calcula total geral

2. LISTAR ESTRATÉGIAS
   Usuário:  strategies
   Sistema:  Lista todas as estratégias registradas
             Mostra status (ATIVA/PARADA)
             Exibe descrição de cada uma

3. PROCESSAMENTO AUTOMÁTICO
   Sistema:  Recebe trades via callback da DLL
             Distribui para todas as estratégias ativas
             Atualiza preços das posições
             Gera logs e alertas


📊 EXEMPLO DE ESTRATÉGIA CUSTOMIZADA
═══════════════════════════════════════════════════════════════════════════════

using ProfitDLLClient.Strategies;

public class MinhaDayTradeStrategy : BaseStrategy
{
    private double _entryPrice;
    private bool _hasPosition;
    
    public override string Name => "Day Trade PETR4";
    public override string Description => "Compra na queda, vende no gain";
    
    public MinhaDayTradeStrategy(PositionManager pm) : base(pm) { }
    
    public override void OnTrade(string ticker, string exchange, 
                                  TConnectorTrade trade)
    {
        base.OnTrade(ticker, exchange, trade);
        
        if (ticker != "PETR4" || exchange != "BOVESPA") return;
        
        var positions = PositionManager.GetPositionsByAsset(ticker, exchange);
        _hasPosition = positions.Count > 0;
        
        if (!_hasPosition && trade.Price < GetLastPrice(ticker, exchange) * 0.995)
        {
            // Caiu 0.5% - sinal de compra!
            Log($"SINAL DE COMPRA: {trade.Price:N2}");
            _entryPrice = trade.Price;
        }
        else if (_hasPosition)
        {
            var position = positions[0];
            if (position.GainLossPercent >= 1.0)
            {
                // Gain de 1% - sinal de venda!
                Log($"SINAL DE VENDA: {trade.Price:N2} | Gain: {position.GainLossPercent:N2}%");
            }
        }
    }
}

// Registrar:
strategyManager.RegisterStrategy(new MinhaDayTradeStrategy(positionManager));


🔧 MÉTODOS ÚTEIS
═══════════════════════════════════════════════════════════════════════════════

PositionManager:
  ├─ UpdatePosition(asset, exchange, account, broker, side, qty, avg, current)
  ├─ UpdatePrice(asset, exchange, price)
  ├─ GetOpenPositions() → List<PositionInfo>
  ├─ GetPositionsByAsset(asset, exchange) → List<PositionInfo>
  ├─ GetPositionsByAccount(accountId, brokerId) → List<PositionInfo>
  ├─ GetTotalGainLoss() → double
  └─ PrintPositionsSummary()

BaseStrategy:
  ├─ Log(message)
  ├─ GetLastPrice(ticker, exchange) → double
  ├─ PositionManager → acesso ao gerenciador
  ├─ IsActive → bool
  └─ LastPrices → Dictionary<string, double>

StrategyManager:
  ├─ RegisterStrategy(IStrategy)
  ├─ StartAll() / StopAll()
  ├─ StartStrategy(name) / StopStrategy(name)
  ├─ ProcessTrade(ticker, exchange, trade)
  ├─ ProcessOfferBookUpdate(ticker, exchange, side, price, qty)
  ├─ ProcessPositionUpdate(accountId, assetId)
  ├─ ProcessOrderUpdate(order)
  ├─ ShowPositions()
  └─ ListStrategies()


⚡ PERFORMANCE
═══════════════════════════════════════════════════════════════════════════════

✅ Lock mínimo: Apenas em operações críticas do PositionManager
✅ Async: Monitoramento em background thread separado
✅ Try-Catch: Erros isolados por estratégia (não derruba outras)
✅ Cache: Últimos preços em memória para acesso rápido
✅ Thread-safe: Dictionary protegido com lock
